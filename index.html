<script>

const $ = (id) => document.getElementById(id);

const esc = (s) => String(s==null?"":s).replace(/[&<>"]/g,(c)=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));

let history = [];

let CODE = sessionStorage.getItem("g4_code") || "";


function showApp(){ $("gate").style.display="none"; $("app").style.display="block"; }

if (CODE) showApp(); else $("gate").style.display="block";


$("enter").addEventListener("click", () => {

  CODE = $("code").value.trim();

  sessionStorage.setItem("g4_code", CODE);

  showApp();

});

$("analisar").addEventListener("click", () => analisar());

$("segmento").addEventListener("keydown", (e) => { if (e.key === "Enter") analisar(); });


async function getPart(segmento, kind) {

  const r = await fetch("/api/briefing", {

    method: "POST",

    headers: { "Content-Type": "application/json", "x-access-code": CODE },

    body: JSON.stringify({ segmento, kind }),

  });

  if (!r.ok) { const e = await r.json().catch(() => ({})); throw new Error(e.error || ("HTTP " + r.status)); }

  return r.json();

}


async function analisar(seg) {

  const target = (seg || $("segmento").value).trim();

  if (!target) return;

  $("segmento").value = target;

  $("analisar").disabled = true;

  $("output").innerHTML = '<div class="status">Analisando mercado e montando o playbook…</div>';


  let mercado=null, playbook=null, err="";

  const [rM, rP] = await Promise.allSettled([getPart(target,"mercado"), getPart(target,"playbook")]);

  if (rM.status==="fulfilled") mercado=rM.value; else err="Mercado: "+rM.reason.message;

  if (rP.status==="fulfilled") playbook=rP.value; else err=(err?err+" | ":"")+"Playbook: "+rP.reason.message;

  $("analisar").disabled = false;


  if (!mercado && !playbook) { $("output").innerHTML = `<div class="alert">Não consegui gerar o briefing.<div class="d">${esc(err)}</div></div>`; return; }

  if (!history.includes(target)) { history=[target,...history].slice(0,6); renderChips(); }

  render({ segmento: target, ...(mercado||{}), ...(playbook||{}) }, err);

}


function renderChips(){

  $("chips").innerHTML = history.map((h)=>`<span class="chip" data-s="${esc(h)}">${esc(h)}</span>`).join("");

  document.querySelectorAll(".chip").forEach((c)=>c.addEventListener("click",()=>analisar(c.dataset.s)));

}

const li = (a)=>(a||[]).map((x)=>`<li>${esc(x)}</li>`).join("");


function render(b, err){

  const card=(c,t,inner,wide)=>`<div class="card${wide?' wide':''}"><h3><span class="dot" style="background:${c}"></span>${t}</h3>${inner}</div>`;

  let h="";

  if (err) h+=`<div class="warn">Parcial — ${esc(err)}</div>`;

  h+=`<div class="bar"><span class="tag">${esc(b.segmento)}</span><button class="copy" id="cp">Copiar briefing</button></div><div class="grid">`;

  if (b.dores_principais) h+=card("#dc2626","Principais dores",`<ul>${li(b.dores_principais)}</ul>`);

  if (b.mercado) h+=card("#059669","Mercado no Brasil",`<ul><li>${esc(b.mercado.resumo)}</li><li><b>${esc(b.mercado.numeros)}</b></li></ul>`);

  if (b.noticias) h+=card("#2563eb","Notícias recentes",`<ul>${(b.noticias||[]).map(n=>`<li><b>${esc(n.titulo)}</b> — ${esc(n.contexto)}</li>`).join("")}</ul>`);

  if (b.tendencias) h+=card("#d97706","Tendências 2026/2027",`<ul>${li(b.tendencias)}</ul>`);

  if (b.perfil_decisor) h+=card("#7c3aed","Perfil do decisor",`<ul><li>${esc(b.perfil_decisor)}</li></ul>`,true);

  if (b.gatilhos) h+=card("#4f46e5","Gatilhos de conexão (G4)",`<ul>${li(b.gatilhos)}</ul>`);

  if (b.perguntas_spin) h+=card("#0d9488","Perguntas de descoberta",`<ul>${li(b.perguntas_spin)}</ul>`);

  if (b.objecoes) h+=card("#db2777","Objeções + contorno",`<ul>${(b.objecoes||[]).map(o=>`<li><b>"${esc(o.objecao)}"</b><br>→ ${esc(o.resposta)}</li>`).join("")}</ul>`,true);

  if (b.vocabulario) h+=card("#6b7280","Vocabulário do setor",`<div class="vocab">${(b.vocabulario||[]).map(v=>`<span>${esc(v)}</span>`).join("")}</div>`,true);

  h+="</div>";

  $("output").innerHTML=h;

  const cp=$("cp");

  if (cp) cp.addEventListener("click",()=>{

    const t=`BRIEFING — ${b.segmento}\n\nDORES\n${(b.dores_principais||[]).map(x=>"- "+x).join("\n")}\n\nMERCADO\n${b.mercado?.resumo||""}\n${b.mercado?.numeros||""}\n\nNOTÍCIAS\n${(b.noticias||[]).map(n=>"- "+n.titulo+" — "+n.contexto).join("\n")}\n\nTENDÊNCIAS\n${(b.tendencias||[]).map(x=>"- "+x).join("\n")}\n\nPERFIL\n${b.perfil_decisor||""}\n\nGATILHOS\n${(b.gatilhos||[]).map(x=>"- "+x).join("\n")}\n\nPERGUNTAS\n${(b.perguntas_spin||[]).map(x=>"- "+x).join("\n")}\n\nOBJEÇÕES\n${(b.objecoes||[]).map(o=>'- "'+o.objecao+'" → '+o.resposta).join("\n")}\n\nVOCABULÁRIO\n${(b.vocabulario||[]).join(", ")}`;

    navigator.clipboard.writeText(t); cp.textContent="Copiado ✓"; setTimeout(()=>cp.textContent="Copiar briefing",1500);

  });

}

</script>
